Here is a Node.js solution. It uses a free exchange rate API to convert your app's USD price into Ghana Cedis (GHS) and calculates the correct subunit (pesewas) before sending the payload to Paystack.
Step 1: Install Axios
You need axios to handle the HTTP requests for both the exchange rate and the Paystack API. Run this in your terminal:
bash
npm install axios
Use code with caution.
Step 2: Implement the Node.js Payment Initializer
Replace your current Paystack initialization logic with this function. It dynamically converts the price so your Paystack account accepts it safely.
javascript
const axios = require('axios');

/**
 * Initializes a Paystack transaction by converting USD to GHS
 * @param {string} email - The customer's email address
 * @param {number} amountInUSD - The price of the item in USD (e.g., 10.50)
 */
async function initializePaystackPayment(email, amountInUSD) {
    try {
        // 1. Fetch live USD to GHS exchange rate
        // (Using a free, no-auth endpoint from exchange-rate.api)
        const rateResponse = await axios.get('https://er-api.com');
        const ghsRate = rateResponse.data.rates.GHS;

        // 2. Convert USD amount to GHS
        const amountInGHS = amountInUSD * ghsRate;

        // 3. Paystack requires amounts in subunits (Pesewas for GHS). 
        // We multiply by 100 and round to prevent floating-point issues.
        const amountInPesewas = Math.round(amountInGHS * 100);

        // 4. Prepare the Paystack Payload forced to GHS
        const paystackPayload = {
            email: email,
            amount: amountInPesewas,
            currency: "GHS", // This matches your Starter account currency
            metadata: {
                original_usd_amount: amountInUSD,
                conversion_rate_used: ghsRate
            }
        };

        // 5. Call Paystack API
        const paystackResponse = await axios.post(
            'https://paystack.co',
            paystackPayload,
            {
                headers: {
                    Authorization: `Bearer YOUR_PAYSTACK_SECRET_KEY`, // Replace with your secret key
                    'Content-Type': 'application/json'
                }
            }
        );

        // 6. Return the authorization URL to redirect your user
        return paystackResponse.data.data.authorization_url;

    } catch (error) {
        console.error("Payment initialization failed:", error.response ? error.response.data : error.message);
        throw error;
    }
}

// initializePaystackPayment('customer@email.com', 25.00)

Step 2: The Backend Code (server.js or index.js) 
Set up an Express.js backend route. It accepts the user's email and the USD price from your frontend, fetches the conversion rate, and handles the Paystack payload internally.
javascript
const express = require('express');
const axios = require('axios');
const app = express();

app.use(express.json());
app.use(express.static('public')); // Serves your frontend files

app.post('/api/checkout', async (req, res) => {
    const { email, usdAmount } = req.body;

    if (!email || !usdAmount) {
        return res.status(400).json({ error: 'Missing email or usdAmount' });
    }

    try {
        // 1. Fetch the live exchange rate
        const rateResponse = await axios.get('https://er-api.com');
        const ghsRate = rateResponse.data.rates.GHS;

        // 2. Convert USD to GHS, then convert GHS to Pesewas (subunits)
        const amountInGHS = usdAmount * ghsRate;
        const amountInPesewas = Math.round(amountInGHS * 100);

        // 3. Create the Paystack payload (Forced to GHS)
        const paystackPayload = {
            email: email,
            amount: amountInPesewas,
            currency: 'GHS',
            metadata: {
                original_usd_price: usdAmount,
                rate_used: ghsRate
            }
        };

        // 4. Hit the Paystack API using the secure Replit environment variable
        const paystackResponse = await axios.post(
            'https://paystack.co',
            paystackPayload,
            {
                headers: {
                    Authorization: `Bearer ${process.env.PAYSTACK_SECRET_KEY}`,
                    'Content-Type': 'application/json'
                }
            }
        );

        // 5. Send the Paystack authorization URL back to your frontend
        res.json({ checkoutUrl: paystackResponse.data.data.authorization_url });

    } catch (error) {
        console.error('Checkout error:', error.response?.data || error.message);
        res.status(500).json({ error: 'Payment initialization failed.' });
    }
});

app.listen(3000, () => console.log('Server running on port 3000'));
Use code with caution.
Step 3: The Frontend Code (public/index.html or script)
Your storefront stays 100% in USD. When the user clicks pay, your script sends the USD value to your backend, gets the payment link back, and redirects them.
javascript
// Example function tied to your USD checkout button
async function handlePayment() {
    const email = document.getElementById('user-email').value;
    const usdAmount = 19.99; // Keep your storefront price in USD

    try {
        const response = await fetch('/api/checkout', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ email, usdAmount })
        });

        const data = await response.json();

        if (data.checkoutUrl) {
            // Redirect the user to the Paystack checkout page
            window.location.href = data.checkoutUrl;
        } else {
            alert('Something went wrong. Please try again.');
        }
    } catch (error) {
        console.error('Frontend payment error:', error);
    }
}
